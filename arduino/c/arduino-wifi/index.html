💻البرمجة: Arduino IDE و mBlock 5

#include <WiFiS3.h>        // UNO R4 WiFi
void setup() {
  Serial.begin(115200);
  WiFi.begin("MyWiFi", "password");
  while (WiFi.status() != WL_CONNECTED) delay(500);
  Serial.println(WiFi.localIP());
}
void loop() {}
